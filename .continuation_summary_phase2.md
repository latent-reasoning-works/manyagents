### Step 2: Modify orchestrator

**File**: `manyagents/main.py`

```python
async def execute_agentic_workflow(goal: str, planner: AgenticPlanner):
    state = {
        'goal': goal,
        'steps_completed': [],
        'current_data': None
    }

    while True:
        next_step = planner.decide_next_step(goal, state)
        if next_step is None:  # Goal achieved
            break

        # Execute step using existing adapter infrastructure
        result = await execute_step(next_step, state)
        state['steps_completed'].append(result)
        state['current_data'] = result.get('embeddings')

    return state
```
